     9→const FUNC_TOKEN = '[func]'
    10→
    11→/** extract the expression stored after "[func]" in the element text */
    12→export function extractFuncExpr(text: string): string {
    13→  const idx = text.indexOf(FUNC_TOKEN)
    14→  if (idx === -1) return ''
    15→  const rest = text.slice(idx + FUNC_TOKEN.length)
    16→  const lineEnd = rest.indexOf('\n')
    17→  return (lineEnd === -1 ? rest : rest.slice(0, lineEnd)).trim()
    18→}

    63→/** build the new element text: replace the segment after the [func] token on its line */
    64→export function withExpr(text: string, expr: string): string {
    65→  const idx = text.indexOf(FUNC_TOKEN)
    66→  const seg = expr.trim() ? ` ${expr.trim()}` : ''
    67→  if (idx === -1) return expr.trim() ? `${text}${text ? '\n' : ''}${FUNC_TOKEN}${seg}`.trimStart() : text
    68→  const lineEnd = text.indexOf('\n', idx)
    69→  const at = idx + FUNC_TOKEN.length
    70→  return text.slice(0, at) + seg + (lineEnd === -1 ? '' : text.slice(lineEnd))
    71→}

    73→/** PNG snapshot with rounded corners + border baked in (corners transparent) */
    74→function snapshotWithFrame(canvas: HTMLCanvasElement): string | null {
    75→  const off = document.createElement('canvas')
    76→  off.width = canvas.width
    77→  off.height = canvas.height
    78→  const ctx = off.getContext('2d')
    79→  if (!ctx) return null
    80→  ctx.drawImage(canvas, 0, 0)
    81→  const cssW = canvas.getBoundingClientRect().width || canvas.width
    82→  const px = canvas.width / cssW // device scale
    83→  const r = 8 * px
    84→  const lw = Math.max(1, Math.round(px))
    85→  ctx.globalCompositeOperation = 'destination-in'
    86→  rrPath(ctx, 0, 0, canvas.width, canvas.height, r)
    87→  ctx.fillStyle = '#ffffff'
    88→  ctx.fill()
    89→  ctx.globalCompositeOperation = 'source-over'
    90→  ctx.strokeStyle = '#d4d4d8'
    91→  ctx.lineWidth = lw
    92→  rrPath(ctx, lw / 2, lw / 2, canvas.width - lw, canvas.height - lw, r)
    93→  ctx.stroke()
    94→  return off.toDataURL('image/png')
    95→}

    97→function rrPath(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
    98→  const rr = Math.max(0, Math.min(r, Math.min(w, h) / 2))
    99→  ctx.beginPath()
   100→  if (typeof ctx.roundRect === 'function') {
   101→    ctx.roundRect(x, y, w, h, rr)
   102→  } else {
   103→    ctx.moveTo(x + rr, y)
   104→    ctx.arcTo(x + w, y, x + w, y + h, rr)
   105→    ctx.arcTo(x + w, y + h, x, y + h, rr)
   106→    ctx.arcTo(x, y + h, x, y, rr)
   107→    ctx.arcTo(x, y, x + w, y, rr)
   108→    ctx.closePath()
   109→  }
   110→}

   112→/** marching squares case table: bit 1=v00, 2=v10, 4=v11, 8=v01; edges 0=bottom,1=right,2=top,3=left */
   113→const CASE_SEGS: Array<Array<[number, number]>> = [
   114→  [],
   115→  [[3, 0]],
   116→  [[0, 1]],
   117→  [[3, 1]],
   118→  [[1, 2]],
   119→  [[3, 2], [0, 1]],
   120→  [[0, 2]],
   121→  [[3, 2]],
   122→  [[2, 3]],
   123→  [[0, 2]],
   124→  [[0, 3], [1, 2]],
   125→  [[1, 2]],
   126→  [[3, 1]],
   127→  [[0, 1]],
   128→  [[0, 3]],
   129→  [],
   130→]

   132→export default function FuncEditor({ el, initial, seed, anchor, viewport, containerRef, stampApiRef, onClose, onWriteText }: FuncEditorProps) {
   133→  const isStamp = initial !== undefined
   134→  const seedSrc = initial ?? seed
   135→  const [expr, setExpr] = useState(() => (initial ? initial.expr : el ? extractFuncExpr(el.text) : ''))
   136→  const [size, setSize] = useState(() => (seedSrc ? { w: seedSrc.w, h: seedSrc.h } : { w: 440, h: 250 }))
   137→  const [view, setView] = useState<FuncView>(() =>
   138→    seedSrc ? { ...seedSrc.view } : { cx: 0, cy: 0, scale: 440 / 13 }
   139→  )
   140→  const inputRef = useRef<HTMLInputElement>(null)
   141→  const canvasRef = useRef<HTMLCanvasElement>(null)
   142→  const rafRef = useRef(0)
   143→  const dragRef = useRef<{ x: number; y: number } | null>(null)
   144→  const resizeRef = useRef<{ x: number; y: number; w: number; h: number } | null>(null)
   145→  const appliedRef = useRef(expr)
   146→  const dirtyRef = useRef(false)
   147→  const exprRef = useRef(expr)
   148→  useEffect(() => {
   149→    exprRef.current = expr
   150→  })
   151→
   152→  const fns = useMemo(() => compileAll(expr), [expr])
   153→  const firstError = fns.find((f) => f.error)?.error ?? null
   154→  const hasAnything = splitExprs(expr).length > 0
   155→
   156→  // focus the expression input on open ("курсор переключается в окошко")
   157→  useEffect(() => {
   158→    const t = setTimeout(() => inputRef.current?.focus(), 30)
   159→    return () => clearTimeout(t)
   160→  }, [])
   161→
   162→  /** (text mode) write expression back into the element text (debounced) */
   163→  useEffect(() => {
   164→    if (isStamp) return
   165→    if (expr === appliedRef.current) return
   166→    const t = setTimeout(() => {
   167→      appliedRef.current = expr
   168→      dirtyRef.current = false
   169→      onWriteText?.(expr)
   170→    }, 400)
   171→    return () => clearTimeout(t)
   172→  }, [expr, isStamp, onWriteText])
   173→
   174→  /** (text mode) final write on close/unmount (expr via ref — the closure would be stale) */
   175→  useEffect(() => {
   176→    if (isStamp) return
   177→    return () => {
   178→      if (dirtyRef.current) {
   179→        dirtyRef.current = false
   180→        onWriteText?.(exprRef.current)
   181→      }
   182→    }
   183→  }, [isStamp, onWriteText])
   184→
   185→  const markDirty = useCallback(() => {
   186→    dirtyRef.current = true
   187→  }, [])
   188→
   189→  // ---- graph drawing -------------------------------------------------------
   190→  const draw = useCallback(() => {
   191→    const canvas = canvasRef.current
   192→    if (!canvas) return
   193→    const dpr = Math.min(window.devicePixelRatio || 1, 2)
   194→    const { w, h } = size
   195→    if (canvas.width !== Math.round(w * dpr) || canvas.height !== Math.round(h * dpr)) {
   196→      canvas.width = Math.round(w * dpr)
   197→      canvas.height = Math.round(h * dpr)
   198→    }
   199→    const ctx = canvas.getContext('2d')
   200→    if (!ctx) return
   201→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   202→    ctx.clearRect(0, 0, w, h)
   203→
   204→    const { cx, cy, scale } = view
   205→    const worldLeft = cx - w / 2 / scale
   206→    const worldRight = cx + w / 2 / scale
   207→    const worldTop = cy + h / 2 / scale
   208→    const worldBottom = cy - h / 2 / scale
   209→    const X = (wx: number) => (wx - cx) * scale + w / 2
   210→    const Y = (wy: number) => h / 2 - (wy - cy) * scale
   211→
   212→    // grid
   213→    const step = niceStep(scale)
   214→    ctx.lineWidth = 1
   215→    ctx.strokeStyle = '#f4f4f5'
   216→    ctx.beginPath()
   217→    for (let gx = Math.ceil(worldLeft / step) * step; gx <= worldRight; gx += step) {
   218→      const px = Math.round(X(gx)) + 0.5
   219→      ctx.moveTo(px, 0)
   220→      ctx.lineTo(px, h)
   221→    }
   222→    for (let gy = Math.ceil(worldBottom / step) * step; gy <= worldTop; gy += step) {
   223→      const py = Math.round(Y(gy)) + 0.5
   224→      ctx.moveTo(0, py)
   225→      ctx.lineTo(w, py)
   226→    }
   227→    ctx.stroke()
   228→
   229→    // axes
   230→    ctx.strokeStyle = '#a1a1aa'
   231→    ctx.beginPath()
   232→    if (Y(0) >= 0 && Y(0) <= h) {
   233→      ctx.moveTo(0, Math.round(Y(0)) + 0.5)
   234→      ctx.lineTo(w, Math.round(Y(0)) + 0.5)
   235→    }
   236→    if (X(0) >= 0 && X(0) <= w) {
   237→      ctx.moveTo(Math.round(X(0)) + 0.5, 0)
   238→      ctx.lineTo(Math.round(X(0)) + 0.5, h)
   239→    }
   240→    ctx.stroke()
   241→
   242→    // tick labels
   243→    ctx.fillStyle = '#71717a'
   244→    ctx.font = '10px ui-sans-serif, system-ui, sans-serif'
   245→    const xAxisY = Math.max(12, Math.min(h - 5, Y(0) + 12))
   246→    for (let gx = Math.ceil(worldLeft / step) * step; gx <= worldRight; gx += step) {
   247→      if (Math.abs(gx) < step / 2) continue
   248→      ctx.fillText(fmtTick(gx, step), X(gx) + 2, xAxisY)
   249→    }
   250→    const yAxisX = Math.max(3, Math.min(w - 34, X(0) + 4))
   251→    for (let gy = Math.ceil(worldBottom / step) * step; gy <= worldTop; gy += step) {
   252→      if (Math.abs(gy) < step / 2) continue
   253→      ctx.fillText(fmtTick(gy, step), yAxisX, Y(gy) - 3)
   254→    }
   255→
   256→    // curves
   257→    const valid = fns.filter((f) => !f.error && f.raw)
   258→    const explicit = valid.filter((f) => !f.usesY)
   259→    const implicit = valid.filter((f) => f.usesY)
   260→
   261→    let N = Math.round(Math.min(Math.max(w, 240), 700))
   262→    if (explicit.length > 1) N = Math.min(N, Math.max(300, Math.floor(3600 / explicit.length)))
   263→    const breakWorld = (h / scale) * 6
   264→
   265→    for (let k = 0; k < explicit.length; k++) {
   266→      const f = explicit[k]
   267→      ctx.beginPath()
   268→      ctx.strokeStyle = COLORS[k % COLORS.length]
   269→      ctx.lineWidth = 2
   270→      ctx.lineJoin = 'round'
   271→      let prevY: number | null = null
   272→      for (let i = 0; i <= N; i++) {
   273→        const wx = worldLeft + ((worldRight - worldLeft) * i) / N
   274→        const y = f.f(wx, 0)
   275→        if (!isFinite(y) || Math.abs(y) > 1e9) {
   276→          prevY = null
   277→          continue
   278→        }
   279→        const sy = Y(y)
   280→        const jump = prevY !== null && Math.abs(y - prevY) > breakWorld
   281→        const signFlip = prevY !== null && prevY * y < 0 && Math.abs(y - prevY) > (h / scale) * 1.5
   282→        if (prevY === null || jump || signFlip) ctx.moveTo(X(wx), sy)
   283→        else ctx.lineTo(X(wx), sy)
   284→        prevY = y
   285→      }
   286→      ctx.stroke()
   287→    }
   288→
   289→    // implicit curves (equations with y) via marching squares
   290→    for (let k = 0; k < implicit.length; k++) {
   291→      const f = implicit[k]
   292→      ctx.beginPath()
   293→      ctx.strokeStyle = COLORS[(explicit.length + k) % COLORS.length]
   294→      ctx.lineWidth = 2
   295→      const gw = 120
   296→      const gh = Math.max(40, Math.round((120 * h) / w))
   297→      const dx = (worldRight - worldLeft) / gw
   298→      const dy = (worldTop - worldBottom) / gh
   299→      const vals = new Float64Array((gw + 1) * (gh + 1))
   300→      for (let j = 0; j <= gh; j++) {
   301→        const wy = worldBottom + j * dy
   302→        for (let i = 0; i <= gw; i++) {
   303→          const v = f.f(worldLeft + i * dx, wy)
   304→          vals[j * (gw + 1) + i] = isFinite(v) && Math.abs(v) < 1e12 ? v : NaN
   305→        }
   306→      }
   307→      for (let j = 0; j < gh; j++) {
   308→        for (let i = 0; i < gw; i++) {
   309→          const v00 = vals[j * (gw + 1) + i]
   310→          const v10 = vals[j * (gw + 1) + i + 1]
   311→          const v11 = vals[(j + 1) * (gw + 1) + i + 1]
   312→          const v01 = vals[(j + 1) * (gw + 1) + i]
   313→          if (isNaN(v00) || isNaN(v10) || isNaN(v11) || isNaN(v01)) continue
   314→          let cs = 0
   315→          if (v00 > 0) cs |= 1
   316→          if (v10 > 0) cs |= 2
   317→          if (v11 > 0) cs |= 4
   318→          if (v01 > 0) cs |= 8
   319→          const segs = CASE_SEGS[cs]
   320→          if (segs.length === 0) continue
   321→          const bx = worldLeft + i * dx
   322→          const by = worldBottom + j * dy
   323→          const pts: Array<[number, number]> = [
   324→            [bx + dx * (v00 / (v00 - v10)), by],
   325→            [bx + dx, by + dy * (v10 / (v10 - v11))],
   326→            [bx + dx * (v01 / (v01 - v11)), by + dy],
   327→            [bx, by + dy * (v00 / (v00 - v01))],
   328→          ]
   329→          for (const [a, b] of segs) {
   330→            ctx.moveTo(X(pts[a][0]), Y(pts[a][1]))
   331→            ctx.lineTo(X(pts[b][0]), Y(pts[b][1]))
   332→          }
   333→        }
   334→      }
   335→      ctx.stroke()
   336→    }
   337→  }, [fns, size, view])
   338→
   339→  // rAF-throttled redraw
   340→  useEffect(() => {
   341→    cancelAnimationFrame(rafRef.current)
   342→    rafRef.current = requestAnimationFrame(draw)
   343→    return () => cancelAnimationFrame(rafRef.current)
   344→  }, [draw])
   345→
   346→  // ---- interactions --------------------------------------------------------
   347→  // wheel zoom around cursor (non-passive to preventDefault)
   348→  useEffect(() => {
   349→    const canvas = canvasRef.current
   350→    if (!canvas) return
   351→    const onWheel = (e: WheelEvent) => {
   352→      e.preventDefault()
   353→      const rect = canvas.getBoundingClientRect()
   354→      const mx = e.clientX - rect.left
   355→      const my = e.clientY - rect.top
   356→      const factor = Math.exp(-e.deltaY * 0.0015)
   357→      setView((v) => {
   358→        const ns = Math.max(2, Math.min(5000, v.scale * factor))
   359→        const wx = v.cx + (mx - size.w / 2) / v.scale
   360→        const wy = v.cy + (size.h / 2 - my) / v.scale
   361→        return { scale: ns, cx: wx - (mx - size.w / 2) / ns, cy: wy - (size.h / 2 - my) / ns }
   362→      })
   363→    }
   364→    canvas.addEventListener('wheel', onWheel, { passive: false })
   365→    return () => canvas.removeEventListener('wheel', onWheel)
   366→  }, [size])
   367→
   368→  const zoomBy = (factor: number) =>
   369→    setView((v) => ({ ...v, scale: Math.max(2, Math.min(5000, v.scale * factor)) }))
   370→
   371→  const resetView = () => setView({ cx: 0, cy: 0, scale: size.w / 13 })
   372→
   373→  const onCanvasPointerDown = (e: React.PointerEvent<HTMLCanvasElement>) => {
   374→    e.preventDefault()
   375→    try {
   376→      ;(e.target as HTMLCanvasElement).setPointerCapture(e.pointerId)
   377→    } catch {
   378→      // pointer may already be released (or synthetic) — the drag still works
   379→    }
   380→    dragRef.current = { x: e.clientX, y: e.clientY }
   381→  }
   382→  const onCanvasPointerMove = (e: React.PointerEvent<HTMLCanvasElement>) => {
   383→    const last = dragRef.current
   384→    if (!last) return
   385→    const dx = e.clientX - last.x
   386→    const dy = e.clientY - last.y
   387→    dragRef.current = { x: e.clientX, y: e.clientY }
   388→    setView((v) => ({ ...v, cx: v.cx - dx / v.scale, cy: v.cy + dy / v.scale }))
   389→  }
   390→  const onCanvasPointerUp = () => {
   391→    dragRef.current = null
   392→  }
   393→
   394→  const onResizePointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
   395→    e.preventDefault()
   396→    e.stopPropagation()
   397→    try {
   398→      ;(e.target as HTMLDivElement).setPointerCapture(e.pointerId)
   399→    } catch {
   400→      // pointer may already be released (or synthetic) — the resize still works
   401→    }
   402→    resizeRef.current = { x: e.clientX, y: e.clientY, w: size.w, h: size.h }
   403→  }
   404→  const onResizePointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
   405→    const st = resizeRef.current
   406→    if (!st) return
   407→    const maxW = Math.min(920, window.innerWidth - 28)
   408→    const maxH = Math.min(640, window.innerHeight - 150)
   409→    setSize({
   410→      w: Math.max(260, Math.min(maxW, st.w + (e.clientX - st.x))),
   411→      h: Math.max(180, Math.min(maxH, st.h + (e.clientY - st.y))),
   412→    })
   413→  }
   414→  const onResizePointerUp = () => {
   415→    resizeRef.current = null
   416→  }
   417→
   418→  // ---- keypad actions (event handlers — ref access is fine here) -----------
   419→  const insert = (str: string, caretBack = 0) => {
   420→    markDirty()
   421→    const input = inputRef.current
   422→    if (!input) {
   423→      setExpr((e) => e + str)
   424→      return
   425→    }
   426→    const s = input.selectionStart ?? input.value.length
   427→    const eEnd = input.selectionEnd ?? s
   428→    const next = input.value.slice(0, s) + str + input.value.slice(eEnd)
   429→    setExpr(next)
   430→    requestAnimationFrame(() => {
   431→      input.focus()
   432→      const pos = s + str.length - caretBack
   433→      input.setSelectionRange(pos, pos)
   434→    })
   435→  }
   436→
   437→  const backspace = () => {
   438→    markDirty()
   439→    const input = inputRef.current
   440→    if (!input) {
   441→      setExpr((e) => e.slice(0, -1))
   442→      return
   443→    }
   444→    const s = input.selectionStart ?? input.value.length
   445→    const eEnd = input.selectionEnd ?? s
   446→    if (s !== eEnd) {
   447→      const next = input.value.slice(0, s) + input.value.slice(eEnd)
   448→      setExpr(next)
   449→      requestAnimationFrame(() => {
   450→        input.focus()
   451→        input.setSelectionRange(s, s)
   452→      })
   453→    } else if (s > 0) {
   454→      const next = input.value.slice(0, s - 1) + input.value.slice(s)
   455→      setExpr(next)
   456→      requestAnimationFrame(() => {
   457→        input.focus()
   458→        input.setSelectionRange(s - 1, s - 1)
   459→      })
   460→    }
   461→  }
   462→
   463→  // ---- panel position ------------------------------------------------------
   464→  const panelW = size.w + 28
   465→  const narrow = viewport.w < 640
   466→  const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v))
   467→  // text mode: panel to the right of the text anchor; stamp mode: cover the stamp
   468→  const left = narrow
   469→    ? clamp((viewport.w - panelW) / 2, 8, Math.max(8, viewport.w - panelW - 10))
   470→    : clamp(initial ? anchor.sx - 13 : anchor.sx + 22, 8, Math.max(8, viewport.w - panelW - 10))
   471→  const top = narrow
   472→    ? 72
   473→    : clamp(initial ? anchor.sy - 96 : anchor.sy - 10, 60, Math.max(60, viewport.h - 200))
   474→
   475→  // ---- stamp API: fresh closure every render (reads left/top/size/view) ----
   476→  useEffect(() => {
   477→    if (!stampApiRef) return
   478→    stampApiRef.current = {
   479→      getStamp: () => {
   480→        const canvas = canvasRef.current
   481→        if (!canvas) return null
   482→        let img: string | null = null
   483→        try {
   484→          img = snapshotWithFrame(canvas)
   485→        } catch {
   486→          img = null
   487→        }
   488→        // canvas offset inside the panel -> panel position is known in viewport coords
   489→        const panel = containerRef.current
   490→        const rect = canvas.getBoundingClientRect()
   491→        let dx = 13
   492→        let dy = 100
   493→        if (panel) {
   494→          const pr = panel.getBoundingClientRect()
   495→          dx = rect.left - pr.left
   496→          dy = rect.top - pr.top
   497→        }
   498→        return {
   499→          expr,
   500→          view: { ...view },
   501→          sx: left + dx,
   502→          sy: top + dy,
   503→          w: size.w,
   504→          h: size.h,
   505→          img,
   506→          valid: fns.some((f) => !f.error && f.raw),
   507→        }
   508→      },
   509→    }
   510→    return () => {
   511→      stampApiRef.current = null
   512→    }
   513→  })
   514→
   515→  return (
   516→    <div
   517→      ref={containerRef}
   518→      role="dialog"
   519→      aria-label={isStamp ? 'Редактирование графика' : 'Редактор функции'}
   520→      data-testid="func-editor"
   521→      className="pointer-events-auto absolute flex flex-col rounded-xl border border-zinc-200 bg-white shadow-2xl"
   522→      style={{ left, top, width: panelW, maxHeight: Math.max(240, viewport.h - top - 16) }}
   523→      onKeyDown={(e) => {
   524→        if (e.key === 'Escape') {
   525→          e.preventDefault()
   526→          e.stopPropagation()
   527→          onClose()
   528→        }
   529→        e.stopPropagation()
   530→      }}
   531→    >
   532→      {/* header */}
   533→      <div className="flex items-center justify-between rounded-t-xl border-b border-zinc-100 bg-zinc-50/80 px-3 py-2">
   534→        <div className="flex items-center gap-2 text-sm font-semibold text-zinc-700">
   535→          <span className="flex h-6 w-6 items-center justify-center rounded-md bg-zinc-800 font-serif italic text-white">ƒ</span>
   536→          {isStamp ? 'Редактирование графика' : 'График функции'}
   537→        </div>
   538→        <div className="flex items-center gap-0.5">
   539→          <button type="button" aria-label="Отдалить график" onClick={() => zoomBy(1 / 1.3)} className="flex h-7 w-7 items-center justify-center rounded-md text-zinc-500 hover:bg-zinc-200/70">
   540→            <Minus className="h-3.5 w-3.5" />
   541→          </button>
   542→          <button type="button" aria-label="Приблизить график" onClick={() => zoomBy(1.3)} className="flex h-7 w-7 items-center justify-center rounded-md text-zinc-500 hover:bg-zinc-200/70">
   543→            <Plus className="h-3.5 w-3.5" />
   544→          </button>
   545→          <button type="button" aria-label="Сбросить вид графика" onClick={resetView} className="flex h-7 w-7 items-center justify-center rounded-md text-zinc-500 hover:bg-zinc-200/70">
   546→            <RotateCcw className="h-3.5 w-3.5" />
   547→          </button>
   548→          <button type="button" aria-label="Закрыть редактор функции" data-testid="func-close" onClick={onClose} className="ml-1 flex h-7 w-7 items-center justify-center rounded-md text-zinc-500 hover:bg-zinc-200/70">
   549→            <X className="h-4 w-4" />
   550→          </button>
   551→        </div>
   552→      </div>
   553→
   554→      <div className="min-h-0 flex-1 overflow-y-auto overflow-x-hidden px-3 pb-3 pt-2">
   555→        {/* expression input */}
   556→        <input
   557→          ref={inputRef}
   558→          data-testid="func-input"
   559→          value={expr}
   560→          onChange={(e) => {
   561→            markDirty()
   562→            setExpr(e.target.value)
   563→          }}
   564→          onKeyDown={(e) => e.stopPropagation()}
   565→          placeholder="sin(x); x^2 + 3x"
   566→          autoComplete="off"
   567→          spellCheck={false}
   568→          aria-label="Выражение функции"
   569→          className="w-full rounded-lg border border-zinc-300 bg-white px-2.5 py-1.5 font-mono text-sm text-zinc-800 outline-none placeholder:text-zinc-400 focus:border-zinc-500"
   570→        />
   571→        {firstError ? (
   572→          <p data-testid="func-error" className="mt-1 text-xs font-medium text-rose-600">
   573→            {firstError}
   574→          </p>
   575→        ) : null}
   576→
   577→        {/* graph */}
   578→        <div className="relative mt-2 overflow-hidden rounded-lg border border-zinc-200" style={{ width: size.w, height: size.h }}>
   579→          <canvas
   580→            ref={canvasRef}
   581→            data-testid="func-canvas"
   582→            width={size.w}
   583→            height={size.h}
   584→            className="block cursor-grab touch-none"
   585→            style={{ width: size.w, height: size.h }}
   586→            onPointerDown={onCanvasPointerDown}
   587→            onPointerMove={onCanvasPointerMove}
   588→            onPointerUp={onCanvasPointerUp}
   589→            onPointerCancel={onCanvasPointerUp}
   590→            aria-label="График функции"
   591→          />
   592→          {!hasAnything ? (
   593→            <div className="pointer-events-none absolute inset-0 flex items-center justify-center">
   594→              <p className="rounded-lg bg-white/85 px-3 py-1.5 text-xs text-zinc-500">Введите функцию — график появится здесь</p>
   595→            </div>
   596→          ) : null}
   597→          {/* resize handle */}
   598→          <div
   599→            data-testid="func-resize"
   600→            onPointerDown={onResizePointerDown}
   601→            onPointerMove={onResizePointerMove}
   602→            onPointerUp={onResizePointerUp}
   603→            onPointerCancel={onResizePointerUp}
   604→            role="separator"
   605→            aria-label="Изменить размер графика"
   606→            className="absolute bottom-0 right-0 flex h-5 w-5 cursor-nwse-resize touch-none items-end justify-end rounded-tl-md bg-gradient-to-br from-transparent to-zinc-300"
   607→            style={{ borderBottomRightRadius: 7 }}
   608→          >
   609→            <svg viewBox="0 0 8 8" className="h-2 w-2 text-zinc-500">
   610→              <path d="M7 1 L1 7 M7 4 L4 7" stroke="currentColor" strokeWidth="1.2" fill="none" />
   611→            </svg>
   612→          </div>
   613→        </div>
   614→
   615→        {/* legend */}
   616→        {fns.length > 0 ? (
   617→          <div className="mt-1.5 flex flex-wrap gap-1.5" data-testid="func-legend">
   618→            {fns.map((f, i) =>
   619→              f.error ? null : (
   620→                <span key={i} className="flex max-w-full items-center gap-1 rounded-full bg-zinc-100 px-2 py-0.5 text-[11px] text-zinc-600">
   621→                  <span className="h-2 w-2 shrink-0 rounded-full" style={{ background: COLORS[colorIndexFor(fns, i) % COLORS.length] }} />
   622→                  <span className="truncate font-mono">{f.raw.length > 26 ? `${f.raw.slice(0, 26)}…` : f.raw}</span>
   623→                </span>
   624→              )
   625→            )}
   626→          </div>
   627→        ) : null}
   628→
   629→        {/* keypad */}
   630→        <div className="mt-2 grid grid-cols-6 gap-1">
   631→          {KEYPAD.map((k) => (
   632→            <button
   633→              key={k.label}
   634→              type="button"
   635→              data-testid={k.testId}
   636→              onMouseDown={(e) => e.preventDefault()}
   637→              onClick={() => (k.del ? backspace() : insert(k.ins ?? '', k.back ?? 0))}
   638→              className="flex h-8 items-center justify-center rounded-md border border-zinc-200 bg-white text-xs font-medium text-zinc-700 hover:bg-zinc-100 active:bg-zinc-200"
   639→            >
   640→              {k.label}
   641→            </button>
   642→          ))}
   643→        </div>
   644→      </div>
   645→    </div>
   646→  )
   647→}
   648→
   649→/** legend color index matching the draw order: explicit curves first, then implicit */
   650→function colorIndexFor(fns: Array<{ usesY: boolean; error: string | null }>, i: number): number {
   651→  let explicitBefore = 0
   652→  let implicitBefore = 0
   653→  for (let k = 0; k < i; k++) {
   654→    if (fns[k].error) continue
   655→    if (fns[k].usesY) implicitBefore++
   656→    else explicitBefore++
   657→  }
   658→  return fns[i].usesY ? explicitBefore + implicitBefore : explicitBefore
   659→}
